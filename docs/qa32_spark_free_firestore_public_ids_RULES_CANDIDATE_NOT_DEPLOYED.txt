// Firestore rules v2 helpers
function sparkOwner(originalId) {
  return signedIn()
    && originalId.matches('^@DEDA-[A-Z0-9]{6}$')
    && exists(/databases/$(database)/documents/users/$(request.auth.uid))
    && exists(/databases/$(database)/documents/deda_sessions/$(request.auth.uid))
    && get(/databases/$(database)/documents/users/$(request.auth.uid))
      .data.sharePersonalId == originalId
    && get(/databases/$(database)/documents/users/$(request.auth.uid))
      .data.accountKey ==
      get(/databases/$(database)/documents/deda_sessions/$(request.auth.uid))
      .data.accountKey
    && exists(/databases/$(database)/documents/deda_share_ids/$(originalId))
    && get(/databases/$(database)/documents/deda_share_ids/$(originalId))
      .data.kind == 'personal'
    && get(/databases/$(database)/documents/deda_share_ids/$(originalId))
      .data.active == true
    && get(/databases/$(database)/documents/deda_share_ids/$(originalId))
      .data.ownerUid == request.auth.uid;
}

function sparkActiveSource(originalId, activeId) {
  return exists(
      /databases/$(database)/documents/deda_stage71_active_identifier/$(originalId))
    && get(
      /databases/$(database)/documents/deda_stage71_active_identifier/$(originalId))
      .data.originalId == originalId
    && get(
      /databases/$(database)/documents/deda_stage71_active_identifier/$(originalId))
      .data.activeId == activeId;
}

function sparkValidGranted(originalId, aliasId, foil) {
  return aliasId.matches('^@DEDA-[A-Z0-9]{4,12}$')
    && foil in ['gold', 'diamond', 'royal', 'pink']
    && exists(
      /databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
    && get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
      .data.identifier == aliasId
    && get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
      .data.recipientOriginalId == originalId
    && get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
      .data.baseColor == foil
    && get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
      .data.group in ['001','002']
    && (
      get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
        .data.kind == 'self_used'
      ||
      (
        get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
          .data.kind == 'gifted'
        && exists(
          /databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId))
        && get(/databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId))
          .data.identifier == aliasId
        && get(/databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId))
          .data.originalId == originalId
        && get(/databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId))
          .data.group ==
            get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId))
              .data.group
        && get(/databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId))
          .data.receivedAt is timestamp
      )
    );
}

match /deda_public_active_aliases/{originalId} {
  allow get: if signedIn();
  allow list, delete: if false;
  allow create, update: if sparkOwner(originalId)
    && request.resource.data.keys().hasOnly(
      ['originalId', 'activeId', 'baseColor', 'updatedAt'])
    && request.resource.data.keys().hasAll(
      ['originalId', 'activeId', 'baseColor', 'updatedAt'])
    && request.resource.data.originalId == originalId
    && request.resource.data.updatedAt == request.time
    && sparkActiveSource(originalId, request.resource.data.activeId)
    && (
      (
        request.resource.data.activeId == originalId
        && request.resource.data.baseColor == null
      )
      ||
      (
        request.resource.data.activeId != originalId
        && sparkValidGranted(originalId,
            request.resource.data.activeId, request.resource.data.baseColor)
      )
    )
    && (!exists(
      /databases/$(database)/documents/deda_public_active_aliases/$(originalId))
      || resource.data.originalId == originalId);
}

match /deda_public_alias_routes/{aliasId} {
  allow get: if signedIn();
  allow list, delete: if false;
  allow create, update: if aliasId.matches('^@DEDA-[A-Z0-9]{4,12}$')
    && request.resource.data.keys().hasOnly(
      ['aliasId', 'originalId', 'baseColor', 'updatedAt'])
    && request.resource.data.keys().hasAll(
      ['aliasId', 'originalId', 'baseColor', 'updatedAt'])
    && request.resource.data.aliasId == aliasId
    && request.resource.data.originalId.matches('^@DEDA-[A-Z0-9]{6}$')
    && request.resource.data.updatedAt == request.time
    && sparkOwner(request.resource.data.originalId)
    && sparkActiveSource(request.resource.data.originalId, aliasId)
    && sparkValidGranted(request.resource.data.originalId,
        aliasId, request.resource.data.baseColor)
    // Reject a false claim if active index was not updated in same logical
    // transaction (getAfter works for existing index on later renewals).
    && getAfter(
      /databases/$(database)/documents/deda_public_active_aliases/$(request.resource.data.originalId))
      .data.activeId == aliasId
    && getAfter(
      /databases/$(database)/documents/deda_public_active_aliases/$(request.resource.data.originalId))
      .data.baseColor == request.resource.data.baseColor
    && (!exists(
      /databases/$(database)/documents/deda_public_alias_routes/$(aliasId))
      || resource.data.originalId == request.resource.data.originalId);
}
