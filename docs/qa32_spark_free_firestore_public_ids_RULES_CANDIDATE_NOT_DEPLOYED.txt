// Firestore rules v2 helpers
// Bundle each source document read just once. This avoids Firestore
// 10-reads-per-operation/20-per-atomic-batch limits on a free Spark plan.
function sparkOwner(originalId) {
  return signedIn() && sparkOwnerAuthed(originalId);
}
function sparkOwnerAuthed(originalId) {
  let account = get(/databases/$(database)/documents/users/$(request.auth.uid));
  let session = get(/databases/$(database)/documents/deda_sessions/$(request.auth.uid));
  let route = get(/databases/$(database)/documents/deda_share_ids/$(originalId));
  return originalId.matches('^@DEDA-[A-Z0-9]{6}$')
    && account.data.sharePersonalId == originalId
    && account.data.accountKey is string
    && account.data.accountKey == session.data.accountKey
    && route.data.publicId == originalId
    && route.data.kind == 'personal'
    && route.data.active == true
    && route.data.ownerUid == request.auth.uid;
}

function sparkActiveSource(originalId, activeId) {
  let current = get(/databases/$(database)/documents/deda_stage71_active_identifier/$(originalId));
  return current.data.originalId == originalId
    && current.data.activeId == activeId;
}

function sparkReceivedGift(originalId, aliasId, group) {
  let receipt = get(/databases/$(database)/documents/deda_stage74_id_gift_receipts/$(aliasId));
  return receipt.data.identifier == aliasId
    && receipt.data.originalId == originalId
    && receipt.data.group == group
    && receipt.data.receivedAt is timestamp;
}

function sparkValidGranted(originalId, aliasId, foil) {
  let grant = get(/databases/$(database)/documents/deda_stage71_granted_registry/$(aliasId));
  return aliasId.matches('^@DEDA-[A-Z0-9]{4,12}$')
    && foil in ['gold','diamond','royal','pink']
    && grant.data.identifier == aliasId
    && grant.data.recipientOriginalId == originalId
    && grant.data.baseColor == foil
    && grant.data.group in ['001','002']
    && (
      grant.data.kind == 'self_used'
      || (grant.data.kind == 'gifted' &&
          sparkReceivedGift(originalId, aliasId, grant.data.group))
    );
}

function sparkProjectedActive(originalId, aliasId, foil) {
  let projection = getAfter(
      /databases/$(database)/documents/deda_public_active_aliases/$(originalId));
  return projection.data.originalId == originalId
    && projection.data.activeId == aliasId
    && projection.data.baseColor == foil;
}

match /deda_public_active_aliases/{originalId} {
  allow get: if signedIn();
  allow list, delete: if false;
  allow create, update: if sparkOwner(originalId)
    && request.resource.data.keys().hasOnly(
      ['originalId', 'activeId', 'baseColor', 'updatedAt'])
    && request.resource.data.keys().hasAll(
      ['originalId', 'activeId', 'baseColor', 'updatedAt'])
    && request.resource.data.originalId == originalId
    && request.resource.data.updatedAt == request.time
    && sparkActiveSource(originalId, request.resource.data.activeId)
    && (
      (
        request.resource.data.activeId == originalId
        && request.resource.data.baseColor == null
      )
      ||
      (
        request.resource.data.activeId != originalId
        && sparkValidGranted(originalId,
            request.resource.data.activeId, request.resource.data.baseColor)
      )
    )
    && (!exists(
      /databases/$(database)/documents/deda_public_active_aliases/$(originalId))
      || resource.data.originalId == originalId);
}

match /deda_public_alias_routes/{aliasId} {
  allow get: if signedIn();
  allow list, delete: if false;
  allow create, update: if aliasId.matches('^@DEDA-[A-Z0-9]{4,12}$')
    && request.resource.data.keys().hasOnly(
      ['aliasId', 'originalId', 'baseColor', 'updatedAt'])
    && request.resource.data.keys().hasAll(
      ['aliasId', 'originalId', 'baseColor', 'updatedAt'])
    && request.resource.data.aliasId == aliasId
    && request.resource.data.originalId.matches('^@DEDA-[A-Z0-9]{6}$')
    && request.resource.data.updatedAt == request.time
    && sparkOwner(request.resource.data.originalId)
    && sparkActiveSource(request.resource.data.originalId, aliasId)
    && sparkValidGranted(request.resource.data.originalId,
        aliasId, request.resource.data.baseColor)
    // A route must match the current index (including this atomic batch).
    && sparkProjectedActive(
      request.resource.data.originalId, aliasId,
      request.resource.data.baseColor)
    && (!exists(
      /databases/$(database)/documents/deda_public_alias_routes/$(aliasId))
      || resource.data.originalId == request.resource.data.originalId);
}
